function zzscs_check_subscription_access() {
    if (!function_exists('wcs_user_has_subscription')) {
        return new WP_Error('zzscs_subscription_system_unavailable', '訂閱驗證系統目前不可用，請聯絡管理員。', ['status' => 503]);
    }

    $uid = get_current_user_id();
    $required_product_id = defined('ZZSCS_REQUIRED_PRODUCT_ID') ? ZZSCS_REQUIRED_PRODUCT_ID : 0; 

    $ok = ($required_product_id > 0) 
        ? (wcs_user_has_subscription($uid, $required_product_id, 'active') || wcs_user_has_subscription($uid, $required_product_id, 'pending-cancel'))
        : (wcs_user_has_subscription($uid, '', 'active') || wcs_user_has_subscription($uid, '', 'pending-cancel'));

    if (!$ok) {
        return new WP_Error('zzscs_no_subscription', 'API 存取被拒：請先訂閱方案。', ['status' => 403]);
    }
    return true;
}

function zzscs_api_permission_check() {
    if (!is_user_logged_in()) return new WP_Error('zzscs_not_logged_in', '請先登入。', ['status' => 401]);

    if (!current_user_can('manage_options')) {
        $sub_check = zzscs_check_subscription_access();
        if (is_wp_error($sub_check)) return $sub_check;
    }

    if (!zzscs_current_user_has_tenant()) return new WP_Error('zzscs_no_tenant', '目前帳號尚未綁定公司資料。', ['status' => 403]);
    
    return true;
}

function zzscs_api_tenant_permission_check() {
    if (!is_user_logged_in()) return new WP_Error('zzscs_not_logged_in', '請先登入。', ['status' => 401]);

    if (!current_user_can('manage_options')) {
        $sub_check = zzscs_check_subscription_access();
        if (is_wp_error($sub_check)) return $sub_check;
    }
    return true;
}

function zzscs_api_add_cash_entry(WP_REST_Request $request) {
    global $wpdb;
    $tenant_id = zzscs_get_current_tenant_id();
    $user_id   = get_current_user_id();

    $date     = sanitize_text_field($request->get_param('date'));
    $memo     = sanitize_text_field($request->get_param('memo'));
    $category = sanitize_text_field($request->get_param('category'));
    $account  = sanitize_text_field($request->get_param('account'));
    $income   = floatval($request->get_param('income'));
    $expense  = floatval($request->get_param('expense'));
    $note     = sanitize_textarea_field($request->get_param('note'));

    if (empty($date)) return new WP_REST_Response(['status' => 'error', 'message' => '請填寫日期。'], 400);
    if (empty($memo)) return new WP_REST_Response(['status' => 'error', 'message' => '請填寫摘要。'], 400);
    if (empty($category)) return new WP_REST_Response(['status' => 'error', 'message' => '請選擇或填寫科目。'], 400);

    if ($income > 0 && $expense > 0) return new WP_REST_Response(['status' => 'error', 'message' => '收入與支出不可同時大於0。'], 400);
    if ($income <= 0 && $expense <= 0) return new WP_REST_Response(['status' => 'error', 'message' => '請輸入大於0的金額。'], 400);

    $table_accounts   = $wpdb->prefix . 'zz_cash_accounts';
    $table_categories = $wpdb->prefix . 'zz_cash_categories';
    $table_entries    = $wpdb->prefix . 'zz_cash_entries';

    $account_row = $wpdb->get_row($wpdb->prepare("SELECT id FROM {$table_accounts} WHERE tenant_id = %d AND account_name = %s AND status = 'active' LIMIT 1", $tenant_id, $account));
    if (!$account_row) return new WP_REST_Response(['status' => 'error', 'message' => '選擇的帳戶不存在或已停用。'], 400);
    $account_id = absint($account_row->id);

    $cat_type = ($income > 0) ? 'income' : 'expense';
    $cat_exists = $wpdb->get_var($wpdb->prepare("SELECT COUNT(*) FROM {$table_categories} WHERE tenant_id = %d AND category_name = %s AND category_type = %s AND status = 'active'", $tenant_id, $category, $cat_type));
    if (!$cat_exists) return new WP_REST_Response(['status' => 'error', 'message' => '請從下拉清單選擇正確的科目，或先至下方「科目管理」新增此科目。'], 400);

    $inserted = $wpdb->insert($table_entries, [
        'tenant_id'  => $tenant_id,
        'entry_date' => $date,
        'memo'       => $memo,
        'category'   => $category,
        'account_id' => $account_id,
        'income'     => $income,
        'expense'    => $expense,
        'note'       => $note,
        'created_by' => $user_id,
        'created_at' => current_time('mysql'),
    ], ['%d', '%s', '%s', '%s', '%d', '%f', '%f', '%s', '%d', '%s']);

    if ($inserted === false) return new WP_REST_Response(['status' => 'error', 'message' => '寫入失敗。'], 500);

    return new WP_REST_Response(['status' => 'ok', 'message' => '新增成功', 'id' => absint($wpdb->insert_id)], 200);
}

function zzscs_api_delete_cash_entry(WP_REST_Request $request) {
    global $wpdb;
    $tenant_id = zzscs_get_current_tenant_id();
    $entry_id  = absint($request->get_param('entry_id'));
    if (!$entry_id) return new WP_REST_Response(['status' => 'error', 'message' => 'entry_id 錯誤。'], 400);

    $table = $wpdb->prefix . 'zz_cash_entries';
    $entry = $wpdb->get_row($wpdb->prepare("SELECT transfer_code FROM {$table} WHERE id = %d AND tenant_id = %d", $entry_id, $tenant_id));
    if (!$entry) return new WP_REST_Response(['status' => 'error', 'message' => '找不到資料。'], 404);

    if (!empty($entry->transfer_code)) {
        $wpdb->query('START TRANSACTION');
        
        $count = $wpdb->get_var($wpdb->prepare("SELECT COUNT(*) FROM {$table} WHERE transfer_code = %s AND tenant_id = %d", $entry->transfer_code, $tenant_id));
        if ($count != 2) {
            $wpdb->query('ROLLBACK');
            return new WP_REST_Response(['status' => 'error', 'message' => '轉帳資料異常（筆數不符），為保護帳本平衡，拒絕刪除。'], 409);
        }

        $deleted = $wpdb->delete($table, ['transfer_code' => $entry->transfer_code, 'tenant_id' => $tenant_id], ['%s', '%d']);
        
        if ($deleted === false || $deleted != 2) {
            $wpdb->query('ROLLBACK');
            return new WP_REST_Response(['status' => 'error', 'message' => '刪除轉帳失敗。'], 500);
        }
        
        $wpdb->query('COMMIT');
    } else {
        $deleted = $wpdb->delete($table, ['id' => $entry_id, 'tenant_id' => $tenant_id], ['%d', '%d']);
        if ($deleted === false || $deleted < 1) {
            return new WP_REST_Response(['status' => 'error', 'message' => '刪除失敗或資料不存在。'], 500);
        }
    }
    
    return new WP_REST_Response(['status' => 'ok', 'message' => '刪除成功'], 200);
}

function zzscs_api_cash_summary() {
    global $wpdb;
    $tenant_id      = zzscs_get_current_tenant_id();
    $table_entries  = $wpdb->prefix . 'zz_cash_entries';
    $table_accounts = $wpdb->prefix . 'zz_cash_accounts';

    $start_of_month = wp_date('Y-m-01');
    $end_of_month   = wp_date('Y-m-t');

    $row = $wpdb->get_row(
        $wpdb->prepare(
            "SELECT 
                COALESCE(SUM(income), 0) AS income_total,
                COALESCE(SUM(expense), 0) AS expense_total,
                COALESCE(SUM(CASE WHEN entry_date >= %s AND entry_date <= %s THEN income ELSE 0 END), 0) AS month_income,
                COALESCE(SUM(CASE WHEN entry_date >= %s AND entry_date <= %s THEN expense ELSE 0 END), 0) AS month_expense
             FROM {$table_entries} 
             WHERE tenant_id = %d 
             AND (transfer_code = '' OR transfer_code IS NULL)", 
             $start_of_month, $end_of_month, 
             $start_of_month, $end_of_month, 
             $tenant_id
        ), ARRAY_A);

    $sql_balance = "SELECT SUM(a.opening_balance + COALESCE(e.total_income, 0) - COALESCE(e.total_expense, 0)) AS real_balance 
                    FROM {$table_accounts} a 
                    LEFT JOIN (SELECT account_id, SUM(income) AS total_income, SUM(expense) AS total_expense 
                               FROM {$table_entries} WHERE tenant_id = %d GROUP BY account_id) e 
                    ON a.id = e.account_id 
                    WHERE a.tenant_id = %d AND a.status = 'active'";
    
    $real_balance = $wpdb->get_var($wpdb->prepare($sql_balance, $tenant_id, $tenant_id));

    return new WP_REST_Response([
        'status'       => 'ok',
        'incomeTotal'  => floatval($row['income_total']),
        'expenseTotal' => floatval($row['expense_total']),
        'balance'      => floatval($real_balance),
        'monthIncome'  => floatval($row['month_income']),
        'monthExpense' => floatval($row['month_expense']),
        'monthBalance' => floatval($row['month_income']) - floatval($row['month_expense']),
    ], 200);
}

function zzscs_api_cash_list(WP_REST_Request $request) {
    global $wpdb;
    $tenant_id      = zzscs_get_current_tenant_id();
    $table_entries  = $wpdb->prefix . 'zz_cash_entries';
    $table_accounts = $wpdb->prefix . 'zz_cash_accounts';

    $limit      = absint($request->get_param('limit'));
    $start_date = sanitize_text_field($request->get_param('start_date'));
    $end_date   = sanitize_text_field($request->get_param('end_date'));
    $keyword    = sanitize_text_field($request->get_param('keyword'));
    $order      = sanitize_text_field($request->get_param('order'));

    if ($limit <= 0 || !in_array($limit, [10, 20, 50, 100], true)) $limit = 10;
    $order_sql = ($order === 'asc') ? 'ASC' : 'DESC';

    $where = "WHERE e.tenant_id = %d";
    $params = [$tenant_id];

    if (!empty($start_date)) { $where .= " AND e.entry_date >= %s"; $params[] = $start_date; }
    if (!empty($end_date)) { $where .= " AND e.entry_date <= %s"; $params[] = $end_date; }
    if (!empty($keyword)) {
        $like = '%' . $wpdb->esc_like($keyword) . '%';
        $where .= " AND (e.memo LIKE %s OR e.category LIKE %s OR a.account_name LIKE %s OR e.note LIKE %s)";
        array_push($params, $like, $like, $like, $like);
    }

    $params[] = $limit;

    $sql = "SELECT e.id, e.entry_date, e.memo, e.category, a.account_name AS account, e.income, e.expense, e.note, e.transfer_code 
            FROM {$table_entries} e LEFT JOIN {$table_accounts} a ON e.account_id = a.id
            {$where} ORDER BY e.entry_date {$order_sql}, e.id {$order_sql} LIMIT %d";

    $rows = $wpdb->get_results($wpdb->prepare($sql, $params), ARRAY_A);
    foreach ($rows as &$row) { $row['income'] = floatval($row['income']); $row['expense'] = floatval($row['expense']); }
    return new WP_REST_Response(['status' => 'ok', 'data' => $rows], 200);
}

    $order_sql = ($order === 'asc') ? 'ASC' : 'DESC';

    $where = "WHERE e.tenant_id = %d";
    $params = [$tenant_id];

    if (!empty($start_date)) { $where .= " AND e.entry_date >= %s"; $params[] = $start_date; }

        array_push($params, $like, $like, $like, $like);
    }

    $sql = "SELECT e.entry_date, e.memo, e.category, a.account_name AS account, e.income, e.expense, e.note 
            FROM {$table_entries} e LEFT JOIN {$table_accounts} a ON e.account_id = a.id
            {$where} ORDER BY e.entry_date {$order_sql}, e.id {$order_sql}";

    $rows = $wpdb->get_results($wpdb->prepare($sql, $params), ARRAY_A);
    $filename = 'cash-export-' . date('Ymd-His') . '.csv';

    nocache_headers();
    header('Content-Type: text/csv; charset=UTF-8');
    header('Content-Disposition: attachment; filename=' . $filename);
    $output = fopen('php://output', 'w');
    fprintf($output, chr(0xEF) . chr(0xBB) . chr(0xBF));
    fputcsv($output, ['日期', '帳戶', '科目', '摘要', '收入', '支出', '備註']);
    foreach ($rows as $row) { fputcsv($output, [$row['entry_date'], $row['account'], $row['category'], $row['memo'], $row['income'], $row['expense'], $row['note']]); }
    fclose($output); exit;
}

function zzscs_api_get_tenants() {

    $user_id = get_current_user_id();

    $table_tenants = $wpdb->prefix . 'zz_tenants';
    $rows = $wpdb->get_results($wpdb->prepare("SELECT t.id, t.company_name FROM {$table_users} tu INNER JOIN {$table_tenants} t ON tu.tenant_id = t.id WHERE tu.user_id = %d AND tu.status = 'active' AND t.status = 'active' ORDER BY t.id ASC", $user_id));

}

function zzscs_api_switch_tenant(WP_REST_Request $request) {
    global $wpdb;
    $user_id = get_current_user_id();
    $tenant_id = absint($request->get_param('tenant_id'));
    if ($tenant_id <= 0) return new WP_REST_Response(['status' => 'error', 'message' => 'tenant_id 不正確。'], 400);
    $table_users = $wpdb->prefix . 'zz_tenant_users';
    $exists = $wpdb->get_var($wpdb->prepare("SELECT COUNT(*) FROM {$table_users} WHERE user_id = %d AND tenant_id = %d AND status = 'active'", $user_id, $tenant_id));

    return new WP_REST_Response(['status' => 'ok', 'message' => '切換成功'], 200);
}

function zzscs_api_create_tenant(WP_REST_Request $request) {
    global $wpdb;
    $user_id = get_current_user_id();
    
    $company_name = sanitize_text_field($request->get_param('company_name'));
    $company_name = preg_replace('/[\pZ\pC]+/u', ' ', $company_name); 
    $company_name = trim($company_name);
    
    if (!$user_id) return new WP_REST_Response(['status' => 'error', 'message' => '請先登入。'], 401);
    if (empty($company_name)) return new WP_REST_Response(['status' => 'error', 'message' => '請填寫公司名稱。'], 400);

    $table_tenants = $wpdb->prefix . 'zz_tenants';
    $table_tenant_users = $wpdb->prefix . 'zz_tenant_users';
    $table_accounts = $wpdb->prefix . 'zz_cash_accounts';
    $table_categories = $wpdb->prefix . 'zz_cash_categories'; 
    $created_at = current_time('mysql');

    $dup = $wpdb->get_var($wpdb->prepare("SELECT COUNT(*) FROM {$table_tenants} WHERE owner_user_id = %d AND company_name = %s", $user_id, $company_name));
    if ($dup) return new WP_REST_Response(['status' => 'error', 'message' => '您已經擁有相同名稱的公司。'], 400);

    $wpdb->query('START TRANSACTION');

function zzscs_api_account_summary() {

    $table_entries  = $wpdb->prefix . 'zz_cash_entries';

    $rows = $wpdb->get_results($wpdb->prepare("SELECT a.id, a.account_name, a.opening_balance, COALESCE(SUM(e.income), 0) AS total_income, COALESCE(SUM(e.expense), 0) AS total_expense FROM {$table_accounts} a LEFT JOIN {$table_entries} e ON a.id = e.account_id AND e.tenant_id = %d WHERE a.tenant_id = %d AND a.status = 'active' GROUP BY a.id, a.account_name, a.opening_balance ORDER BY a.account_name ASC", $tenant_id, $tenant_id), ARRAY_A);

    $data = [];
    foreach ($rows as $row) {
        $opening = floatval($row['opening_balance']); $income = floatval($row['total_income']); $expense = floatval($row['total_expense']);

    }
    return new WP_REST_Response(['status' => 'ok', 'data' => $data], 200);
}

function zzscs_api_transfer_create(WP_REST_Request $request) {
    global $wpdb;
    $tenant_id = zzscs_get_current_tenant_id();
    $user_id   = get_current_user_id();

    $date         = sanitize_text_field($request->get_param('date'));
    $from_account = sanitize_text_field($request->get_param('from_account'));
    $to_account   = sanitize_text_field($request->get_param('to_account'));
    $amount       = floatval($request->get_param('amount'));
    $note         = sanitize_textarea_field($request->get_param('note'));

    if (empty($date)) return new WP_REST_Response(['status' => 'error', 'message' => '請填寫轉帳日期。'], 400);
    if (empty($from_account)) return new WP_REST_Response(['status' => 'error', 'message' => '請選擇轉出帳戶。'], 400);
    if (empty($to_account)) return new WP_REST_Response(['status' => 'error', 'message' => '請選擇轉入帳戶。'], 400);
    if ($from_account === $to_account) return new WP_REST_Response(['status' => 'error', 'message' => '轉出帳戶與轉入帳戶不可相同。'], 400);
    if ($amount <= 0) return new WP_REST_Response(['status' => 'error', 'message' => '轉帳金額必須大於0。'], 400);

    $table_accounts = $wpdb->prefix . 'zz_cash_accounts';
    $table_entries  = $wpdb->prefix . 'zz_cash_entries';

    $from_row = $wpdb->get_row($wpdb->prepare("SELECT id FROM {$table_accounts} WHERE tenant_id = %d AND account_name = %s AND status = 'active' LIMIT 1", $tenant_id, $from_account));
    if (!$from_row) return new WP_REST_Response(['status' => 'error', 'message' => '找不到轉出帳戶，或帳戶已停用。'], 404);

    $to_row = $wpdb->get_row($wpdb->prepare("SELECT id FROM {$table_accounts} WHERE tenant_id = %d AND account_name = %s AND status = 'active' LIMIT 1", $tenant_id, $to_account));
    if (!$to_row) return new WP_REST_Response(['status' => 'error', 'message' => '找不到轉入帳戶，或帳戶已停用。'], 404);

    $transfer_code = 'TRF' . date('YmdHis') . wp_rand(100, 999);
    $created_at    = current_time('mysql');
    $memo = '帳戶轉帳';

    $wpdb->query('START TRANSACTION');

    $result1 = $wpdb->insert($table_entries, [
        'tenant_id' => $tenant_id, 'entry_date' => $date, 'memo' => $memo, 'category' => '帳戶轉帳', 
        'account_id' => absint($from_row->id), 'income' => 0, 'expense' => $amount, 'note' => $note, 
        'transfer_code' => $transfer_code, 'created_by' => $user_id, 'created_at' => $created_at
    ], ['%d', '%s', '%s', '%s', '%d', '%f', '%f', '%s', '%s', '%d', '%s']);
    
    $result2 = $wpdb->insert($table_entries, [
        'tenant_id' => $tenant_id, 'entry_date' => $date, 'memo' => $memo, 'category' => '帳戶轉帳', 
        'account_id' => absint($to_row->id), 'income' => $amount, 'expense' => 0, 'note' => $note, 
        'transfer_code' => $transfer_code, 'created_by' => $user_id, 'created_at' => $created_at
    ], ['%d', '%s', '%s', '%s', '%d', '%f', '%f', '%s', '%s', '%d', '%s']);

    if ($result1 === false || $result2 === false) {
        $wpdb->query('ROLLBACK');
        return new WP_REST_Response(['status' => 'error', 'message' => '建立轉帳資料失敗，已還原。'], 500);
    } else {
        $wpdb->query('COMMIT');
        return new WP_REST_Response(['status' => 'ok', 'message' => '轉帳成功', 'transfer_code' => $transfer_code], 200);
    }
}

// 當訂單狀態變更為 processing 或 completed 時，觸發檢查
add_action('woocommerce_order_status_processing', 'zzscs_auto_create_tenant_on_purchase', 10, 1);
add_action('woocommerce_order_status_completed', 'zzscs_auto_create_tenant_on_purchase', 10, 1);

function zzscs_auto_create_tenant_on_purchase($order_id) {
    $order = wc_get_order($order_id);
    
    // 🔴 優化一：加入嚴格的 Order 物件型別檢查
    if (!$order || !is_a($order, 'WC_Order')) {
        return;
    }

    // 1. 為了避免重複執行，檢查是否已經成功建立過 (yes) 或已存在 (exists)
    $meta_status = $order->get_meta('_zzscs_tenant_created');
    if (in_array($meta_status, ['yes', 'exists'], true)) {
        return;
    }

    $user_id = $order->get_user_id();
    if (!$user_id) return; // 訪客購買不處理

    // 2. 取得我們定義的專屬商品 ID，並轉為整數
    $required_product_id = (int) (defined('ZZSCS_REQUIRED_PRODUCT_ID') ? ZZSCS_REQUIRED_PRODUCT_ID : 68568);
    $has_subscription = false;

    // 🔴 優化二：商品 ID 嚴格轉型與比對 (===)
    foreach ($order->get_items() as $item) {
        $product_id   = (int) $item->get_product_id();
        $variation_id = (int) $item->get_variation_id();
        
        if ($product_id === $required_product_id || $variation_id === $required_product_id) {
            $has_subscription = true;
            break;
        }
    }

    // 3. 非記帳訂單直接「靜默離開」，保持後台乾淨
    if (!$has_subscription) {
        return;
    }

    // 接下來是確認購買記帳方案的流程
    global $wpdb;
    
    $table_tenants         = $wpdb->prefix . 'zz_tenants';
    $table_tenant_users    = $wpdb->prefix . 'zz_tenant_users';
    $table_cash_accounts   = $wpdb->prefix . 'zz_cash_accounts';
    $table_cash_categories = $wpdb->prefix . 'zz_cash_categories';

    // 4. 檢查客戶是否已有公司帳本 (同時補回遺失的 Meta)
    $existing_tenant_id = get_user_meta($user_id, 'zz_default_tenant_id', true);

    if (empty($existing_tenant_id)) {
        $existing_tenant_id = $wpdb->get_var($wpdb->prepare(
            "SELECT id FROM {$table_tenants} WHERE owner_user_id = %d AND status = 'active' ORDER BY id ASC LIMIT 1",
            $user_id
        ));
    }

    if (!empty($existing_tenant_id)) {
        update_user_meta($user_id, 'zz_default_tenant_id', $existing_tenant_id);
        // 保險起見，舊客戶也一併刷新狀態
        update_user_meta($user_id, 'zz_saas_plan', 'simple-cash-saas');
        update_user_meta($user_id, 'zz_saas_status', 'active');
        
        $order->add_order_note('贊贊記帳：客戶已擁有公司帳本，略過自動建立。tenant_id=' . $existing_tenant_id);
        $order->update_meta_data('_zzscs_tenant_created', 'exists');
        $order->save();
        return;
    }

    // 5. 動態產生友善的帳本名稱
    $billing_company   = sanitize_text_field(trim($order->get_billing_company()));
    $billing_lastname  = sanitize_text_field(trim($order->get_billing_last_name()));
    $billing_firstname = sanitize_text_field(trim($order->get_billing_first_name()));
    $full_name         = trim($billing_lastname . $billing_firstname);

    if (!empty($billing_company)) {
        $company_name = $billing_company;
    } elseif (!empty($full_name)) {
        $company_name = $full_name . '的帳本';
    } else {
        $company_name = '我的記帳本';
    }

    $created_at = current_time('mysql');

    // ==========================================
    // 6. 開始資料庫寫入交易 (Transaction)
    // ==========================================
    $wpdb->query('START TRANSACTION');

    $insert_tenant = $wpdb->insert($table_tenants, [
        'company_name'  => $company_name,
        'owner_user_id' => $user_id,
        'plan_code'     => 'simple-cash-saas',
        'status'        => 'active',
        'created_at'    => $created_at
    ], ['%s', '%d', '%s', '%s', '%s']);

    if ($insert_tenant === false) {
        $wpdb->query('ROLLBACK');
        $order->add_order_note('贊贊記帳：自動建立公司失敗 (DB Insert Error)。');
        return;
    }

    $tenant_id = absint($wpdb->insert_id);

    // 寫入預設關聯資料
    $i_user = $wpdb->insert($table_tenant_users, ['tenant_id' => $tenant_id, 'user_id' => $user_id, 'tenant_role' => 'owner', 'status' => 'active', 'created_at' => $created_at], ['%d', '%d', '%s', '%s', '%s']);
    $i_acc  = $wpdb->insert($table_cash_accounts, ['tenant_id' => $tenant_id, 'account_name' => '現金', 'account_type' => 'cash', 'opening_balance' => 0, 'status' => 'active', 'created_at' => $created_at], ['%d', '%s', '%s', '%f', '%s', '%s']);
    
    $i_c1 = $wpdb->insert($table_cash_categories, ['tenant_id' => $tenant_id, 'category_name' => '餐費', 'category_type' => 'expense', 'status' => 'active', 'created_at' => $created_at], ['%d', '%s', '%s', '%s', '%s']);
    $i_c2 = $wpdb->insert($table_cash_categories, ['tenant_id' => $tenant_id, 'category_name' => '交通', 'category_type' => 'expense', 'status' => 'active', 'created_at' => $created_at], ['%d', '%s', '%s', '%s', '%s']);
    $i_c3 = $wpdb->insert($table_cash_categories, ['tenant_id' => $tenant_id, 'category_name' => '住宿', 'category_type' => 'expense', 'status' => 'active', 'created_at' => $created_at], ['%d', '%s', '%s', '%s', '%s']);
    $i_c4 = $wpdb->insert($table_cash_categories, ['tenant_id' => $tenant_id, 'category_name' => '一般收入', 'category_type' => 'income', 'status' => 'active', 'created_at' => $created_at], ['%d', '%s', '%s', '%s', '%s']);

    // 🔴 修正：拿掉多餘的 DELETE，信任 ROLLBACK 機制
    if ($i_user === false || $i_acc === false || $i_c1 === false || $i_c2 === false || $i_c3 === false || $i_c4 === false) {
        $wpdb->query('ROLLBACK');
        $order->add_order_note('贊贊記帳：初始化公司預設資料失敗，已還原。');
        return;
    }

    $wpdb->query('COMMIT');
    
    // 🔴 優化三：寫入完整的 User Meta，方便未來營運管理
    update_user_meta($user_id, 'zz_default_tenant_id', $tenant_id);
    update_user_meta($user_id, 'zz_saas_plan', 'simple-cash-saas');
    update_user_meta($user_id, 'zz_saas_status', 'active');

    // 標記為已建立 (yes)，防重複執行
    $order->update_meta_data('_zzscs_tenant_created', 'yes');
    $order->save();

    // 留存備註方便客服追蹤
    $order->add_order_note("贊贊記帳：已自動為客戶建立預設帳本 (Tenant ID: {$tenant_id}，名稱：{$company_name})。");
}

      limit: document.getElementById("zz-filter-limit") ? document.getElementById("zz-filter-limit").value : "10",
      order: document.getElementById("zz-filter-order") ? document.getElementById("zz-filter-order").value : "desc"
    };
  }

  function buildListUrl() {
    var filters = getFilters();
    var params = new URLSearchParams();
    if (filters.start_date) params.append("start_date", filters.start_date);
    if (filters.end_date) params.append("end_date", filters.end_date);
    if (filters.keyword) params.append("keyword", filters.keyword);
    if (filters.limit) params.append("limit", filters.limit);
    if (filters.order) params.append("order", filters.order);
    return ZZSCS_DATA.apiBase + "/cash-list?" + params.toString();
  }

  function buildExportUrl() {
    var filters = getFilters();
    var params = new URLSearchParams();
    if (filters.start_date) params.append("start_date", filters.start_date);
    if (filters.end_date) params.append("end_date", filters.end_date);
    if (filters.keyword) params.append("keyword", filters.keyword);
    if (filters.order) params.append("order", filters.order);
    params.append("_wpnonce", ZZSCS_DATA.nonce);
    return ZZSCS_DATA.apiBase + "/cash-export?" + params.toString();
  }

        opt.textContent = t.name || t.company_name || ("公司#" + t.id);

        select.appendChild(opt);

  async function addCashRecord() {
    var type = document.getElementById("zz-type").value;
    var amount = parseFloat(document.getElementById("zz-amount").value) || 0;
    
    var data = {
      date: document.getElementById("zz-date").value,
      memo: document.getElementById("zz-memo").value,
      category: document.getElementById("zz-category").value,
      account: document.getElementById("zz-account").value,
      income: type === 'income' ? amount : 0,
      expense: type === 'expense' ? amount : 0,
      note: document.getElementById("zz-note").value
    };

    if (!data.date) { showMsg("請先填寫日期", true); return; }
    if (!data.account) { showMsg("請選擇帳戶", true); return; }
    if (!data.category) { showMsg("請填寫科目", true); return; }
    if (amount <= 0) { showMsg("請輸入大於0的金額", true); return; }
    if (!data.memo) { showMsg("請先填寫摘要", true); return; }

    try {
      setLoading(true, "送出記帳中...");
      clearAllMsgs();
      var res = await fetch(ZZSCS_DATA.apiBase + "/cash-entry", { method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json", "X-WP-Nonce": ZZSCS_DATA.nonce }, body: JSON.stringify(data) });
      var json = await res.json();
      if (json.status === "ok") {
        clearForm();
        showMsg("記帳成功，編號：" + json.id, false); 
        await loadCategories(); await loadAccounts(); await loadSummary(); await loadAccountSummary(); await loadAccountManageList(); await loadList();
      } else { showMsg(json.message || "新增失敗", true); }
    } catch (err) { showMsg("API連線失敗：" + err, true); } 
    finally { setLoading(false, ""); }
  }

  async function loadAccountManageList() {
    var tbody = document.getElementById("zz-account-manage-tbody");
    if (!tbody) return;
    tbody.innerHTML = '<tr><td colspan="5">讀取中...</td></tr>';
    try {
      var res = await fetch(ZZSCS_DATA.apiBase + "/account-list", { method: "GET", credentials: "same-origin", headers: { "X-WP-Nonce": ZZSCS_DATA.nonce } });

        if (!json.data || json.data.length === 0) { tbody.innerHTML = '<tr><td colspan="5">目前沒有帳戶資料</td></tr>'; return; }

          var actionHtml = item.status === "active" ?           var typeText = item.account_type === "bank" ? "銀行" : (item.account_type === "cash" ? "現金" : "其他");

  async function createTransfer() {

  async function loadList() {
    var tbody = document.getElementById("zz-tbody");
    if (!tbody) return;
    tbody.innerHTML = '<tr><td colspan="8">讀取中...</td></tr>';
    try {
      var res = await fetch(buildListUrl(), { method: "GET", credentials: "same-origin", headers: { "X-WP-Nonce": ZZSCS_DATA.nonce } });
      var json = await res.json();
      if (json.status === "ok") {
        if (!json.data || json.data.length === 0) { tbody.innerHTML = '<tr><td colspan="8">目前沒有資料</td></tr>'; return; }
        tbody.innerHTML = json.data.map(function (item) {
          var categoryDisplay = escapeHtml(item.category);
          var actionHtml = '';
          
          // 🔴 關鍵更新：將轉帳屬性附加在 dataset 中，供後續精準判斷
          if (item.transfer_code) {
              categoryDisplay += ' <span class="zz-transfer-badge">轉帳</span>';
              actionHtml = '<button type="button" class="zz-entry-delete" data-id="' + item.id + '" data-istransfer="1">刪除轉帳(整組)</button>';
          } else {
              actionHtml = '<button type="button" class="zz-entry-edit" data-id="' + item.id + '">編輯</button> ' +
                           '<button type="button" class="zz-entry-delete" data-id="' + item.id + '" data-istransfer="0">刪除</button>';
          }

          return "<tr><td>" + escapeHtml(item.entry_date) + "</td><td>" + escapeHtml(item.account) + "</td><td>" + categoryDisplay + "</td><td>" + escapeHtml(item.memo) + "</td><td>" + formatNumber(item.income) + "</td><td>" + formatNumber(item.expense) + "</td><td>" + escapeHtml(item.note) + "</td><td>" + actionHtml + "</td></tr>";
        }).join("");
        
        document.querySelectorAll(".zz-entry-edit").forEach(function (btn) { btn.addEventListener("click", function () { var id = this.dataset.id; var target = json.data.find(function (row) { return String(row.id) === String(id); }); if (target) setEditMode(target); }); });
        
        // 🔴 傳遞 isTransfer 狀態給 delete 函式
        document.querySelectorAll(".zz-entry-delete").forEach(function (btn) { 
            btn.addEventListener("click", function () { 
                var isTransfer = this.dataset.istransfer === "1";
                deleteCashRecord(this.dataset.id, isTransfer); 
            }); 
        });
      } else { tbody.innerHTML = '<tr><td colspan="8">讀取失敗</td></tr>'; }
    } catch (err) { tbody.innerHTML = '<tr><td colspan="8">API連線失敗</td></tr>'; console.error("loadList error:", err); }
  }
